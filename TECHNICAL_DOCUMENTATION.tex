\section{Executive Summary}

This document provides comprehensive technical documentation for the Smart Home Monitoring \& Control System developed as part of SCS 3311. The system consists of a React Native mobile application with complete UI/UX implementation, designed for cloud-based backend integration and hardware simulation.

\subsection{Project Status}
\begin{itemize}[leftmargin=*]
    \item \textcolor{implemented}{\textbf{COMPLETED:}} Full mobile application frontend with all required UI/UX components
    \item \textcolor{pending}{\textbf{IN PROGRESS:}} Backend infrastructure and cloud database implementation
    \item \textcolor{required}{\textbf{REQUIRED:}} Hardware simulator dashboard and real-time synchronization
\end{itemize}

\subsection{Technology Stack}
\begin{itemize}[leftmargin=*]
    \item \textbf{Mobile Framework:} React Native with Expo SDK v54
    \item \textbf{Navigation:} Expo Router (file-based routing)
    \item \textbf{UI Components:} Custom glassmorphism design system
    \item \textbf{Planned Backend:} Supabase (PostgreSQL + Real-time subscriptions)
    \item \textbf{State Management:} Local mock data (to be replaced with cloud sync)
\end{itemize}

\textbf{Example Current Implementation (Mock):}
\begin{verbatim}
export async function getFloors(): Promise<Floor[]> {
  return Promise.resolve(FLOORS); // In-memory array
}
\end{verbatim}

\textbf{Required Implementation (Supabase):}
\begin{verbatim}
export async function getFloors(): Promise<Floor[]> {
  const { data, error } = await supabase
    .from('floors')
    .select('*')
    .order('level', { ascending: true });
  if (error) throw error;
  return data;
}
\end{verbatim}

\section{Database Schema Design}

\subsection{Proposed Supabase Tables}


\subsubsection{floors Table}
\begin{verbatim}
CREATE TABLE floors (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  level INTEGER NOT NULL,
  device_count INTEGER DEFAULT 0,
  active_device_count INTEGER DEFAULT 0,
  floor_plan_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
\end{verbatim}

\subsubsection{devices Table}
\begin{verbatim}
CREATE TABLE devices (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  floor_id UUID REFERENCES floors(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  type TEXT NOT NULL CHECK (type IN ('light', 'thermostat', 
    'lock', 'camera', 'fan', 'tv', 'speaker', 'outlet', 
    'iron', 'multiSwitch')),
  status TEXT NOT NULL DEFAULT 'off' CHECK (status IN 
    ('on', 'off', 'error', 'offline')),
  room_name TEXT,
  value NUMERIC,
  unit TEXT,
  icon_name TEXT,
  safety_timeout INTEGER,
  last_updated TIMESTAMPTZ DEFAULT NOW(),
  created_at TIMESTAMPTZ DEFAULT NOW()
);
\end{verbatim}

\subsubsection{switch\_circuits Table}
\begin{verbatim}
CREATE TABLE switch_circuits (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  device_id UUID REFERENCES devices(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'off' CHECK (status IN 
    ('on', 'off', 'error', 'disconnected')),
  power NUMERIC,
  position INTEGER NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
\end{verbatim}


\subsubsection{schedules Table}
\begin{verbatim}
CREATE TABLE schedules (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  device_id UUID REFERENCES devices(id) ON DELETE CASCADE,
  type TEXT NOT NULL CHECK (type IN ('time', 'safety')),
  enabled BOOLEAN DEFAULT true,
  time TIME,
  action TEXT CHECK (action IN ('on', 'off')),
  days INTEGER[],
  max_duration_minutes INTEGER,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
\end{verbatim}

\subsubsection{floor\_plan\_pins Table}
\begin{verbatim}
CREATE TABLE floor_plan_pins (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  floor_id UUID REFERENCES floors(id) ON DELETE CASCADE,
  device_id UUID REFERENCES devices(id) ON DELETE CASCADE,
  x NUMERIC NOT NULL,
  y NUMERIC NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(floor_id, device_id)
);
\end{verbatim}

\subsubsection{alerts Table}
\begin{verbatim}
CREATE TABLE alerts (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  device_id UUID REFERENCES devices(id) ON DELETE SET NULL,
  type TEXT NOT NULL CHECK (type IN ('safety', 'error', 
    'offline', 'info')),
  title TEXT NOT NULL,
  message TEXT,
  read BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
\end{verbatim}

\subsection{Row-Level Security Policies}
All tables must implement RLS policies to ensure users only access their own data:

\begin{verbatim}
ALTER TABLE floors ENABLE ROW LEVEL SECURITY;
ALTER TABLE devices ENABLE ROW LEVEL SECURITY;
-- ... enable for all tables

CREATE POLICY "Users can view their own floors"
  ON floors FOR SELECT
  USING (auth.uid() = user_id);

CREATE POLICY "Users can insert their own floors"
  ON floors FOR INSERT
  WITH CHECK (auth.uid() = user_id);
-- ... similar policies for UPDATE and DELETE
\end{verbatim}

import { supabase } from './supabase';

// Replace mock implementation with real database operations
export async function toggleDevice(id: string): 
    Promise<Device | undefined> {
  // Fetch current device state
  const { data: device, error: fetchError } = 
    await supabase
      .from('devices')
      .select('*')
      .eq('id', id)
      .single();

  if (fetchError || !device) return undefined;

  // Toggle status
  const newStatus = device.status === 'on' ? 'off' : 'on';

  // Update in database
  const { data: updated, error: updateError } = 
    await supabase
      .from('devices')
      .update({ 
        status: newStatus, 
        last_updated: new Date().toISOString() 
      })
      .eq('id', id)
      .select()
      .single();

  if (updateError) throw updateError;
  return updated;
}
\end{verbatim}

import { useEffect, useState } from 'react';
import { supabase } from '@/services/supabase';

function HomeScreen() {
  const [devices, setDevices] = useState<Device[]>([]);

  useEffect(() => {
    // Initial data fetch
    fetchDevices();

    // Subscribe to real-time changes
    const subscription = supabase
      .channel('devices-channel')
      .on('postgres_changes', 
        { 
          event: '*', 
          schema: 'public', 
          table: 'devices' 
        },
        (payload) => {
          console.log('Device changed:', payload);
          // Refresh device list on any change
          fetchDevices();
        }
      )
      .subscribe();

    // Cleanup subscription
    return () => {
      subscription.unsubscribe();
    };
  }, []);

  async function fetchDevices() {
    const { data, error } = await supabase
      .from('devices')
      .select('*')
      .order('last_updated', { ascending: false });
    
    if (!error && data) {
      setDevices(data);
    }
  }

  return (
    // ... UI components
  );
}
\end{verbatim}

\subsection{Hardware Simulator Example}

\textbf{Technology:} React.js + Supabase + Tailwind CSS

\textbf{File:} \texttt{simulator/src/DeviceSimulator.jsx}

\begin{verbatim}
import { useEffect, useState } from 'react';
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  process.env.REACT_APP_SUPABASE_URL,
  process.env.REACT_APP_SUPABASE_ANON_KEY
);

function DeviceSimulator() {
  const [devices, setDevices] = useState([]);

  useEffect(() => {
    // Fetch initial devices
    fetchDevices();

    // Listen to real-time changes
    const subscription = supabase
      .channel('simulator-channel')
      .on('postgres_changes', {
        event: '*',
        schema: 'public',
        table: 'devices'
      }, (payload) => {
        console.log('Device update:', payload);
        fetchDevices(); // Refresh display
      })
      .subscribe();

    return () => subscription.unsubscribe();
  }, []);

  async function fetchDevices() {
    const { data } = await supabase
      .from('devices')
      .select('*');
    setDevices(data || []);
  }

  return (
    <div className="p-8">
      <h1 className="text-3xl font-bold mb-6">
        Hardware Simulator
      </h1>
      <div className="grid grid-cols-4 gap-4">
        {devices.map(device => (
          <DeviceCard key={device.id} device={device} />
        ))}
      </div>
    </div>
  );
}

function DeviceCard({ device }) {
  const isOn = device.status === 'on';
  
  return (
    <div className={`p-4 rounded-lg border-2 ${
      isOn ? 'bg-green-100 border-green-500' 
           : 'bg-gray-100 border-gray-300'
    }`}>
      <div className="flex items-center gap-2">
        <div className={`w-4 h-4 rounded-full ${
          isOn ? 'bg-green-500' : 'bg-gray-400'
        }`} />
        <span className="font-semibold">{device.name}</span>
      </div>
      <p className="text-sm text-gray-600 mt-2">
        {device.room_name}
      </p>
      <p className="text-xs text-gray-500 mt-1">
        Status: {device.status.toUpperCase()}
      </p>
    </div>
  );
}

export default DeviceSimulator;
\end{verbatim}
